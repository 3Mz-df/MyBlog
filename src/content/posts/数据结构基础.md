## 单链表
#include <stdio.h>

struct Node {
    int data;
    struct Node *next;
};

int main(void) {
    // 创建三个节点
    struct Node a, b, c;

    // 给节点存入数据
    a.data = 10;
    b.data = 20;
    c.data = 30;

    // 用指针把节点连接起来
    a.next = &b;
    b.next = &c;
    c.next = NULL;

    // head 保存第一个节点的地址
    struct Node *head = &a;

    // 从头开始，依次访问每个节点
    struct Node *p = head;

    while (p != NULL) {
        printf("%d -> ", p->data);
        p = p->next;
    }

    printf("NULL\n");

    return 0;
}
输出：10 -> 20 -> 30 -> NULL
- 对于节点abc
    是结构体变量，里面有 data 和 next
     data：存整数  
     next：存地址 
     a.data = 10;  // 把 10 放进 a 的 data 格子
     创建节点时就给它赋初值，可以写：struct Node a = {10, NULL};
- &b	获取节点 b 的地址
- a.next = &b	让节点 a 指向节点 b
- p->data	访问指针 p 所指节点的数据
- "%d -> " 是 printf 的输出格式字符串，可以拆成两部分：
    %d：占位符，表示这里输出一个整数。
    ->：普通文字，会原样输出空格和箭头。
