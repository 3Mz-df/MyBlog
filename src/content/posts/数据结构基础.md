## 单链表
#include <stdio.h>

struct Node {
    int data;
    struct Node *next;
};

int main(void) {
    // 创建三个节点
    struct Node a, b, c;

    // 给节点存入数据
    a.data = 10;
    b.data = 20;
    c.data = 30;

    // 用指针把节点连接起来
    a.next = &b;
    b.next = &c;
    c.next = NULL;

    // head 保存第一个节点的地址
    struct Node *head = &a;

    // 从头开始，依次访问每个节点
    struct Node *p = head;

    while (p != NULL) {
        printf("%d -> ", p->data);
        p = p->next;
    }

    printf("NULL\n");

    return 0;
}
输出：10 -> 20 -> 30 -> NULL

- 对于节点abc
是结构体变量，里面有 data 和 next
data：存整数  
next：存地址 
a.data = 10;  // 把 10 放进 a 的 data 格子
创建节点时就给它赋初值，可以写：struct Node a = {10, NULL};

- &b	
获取节点 b 的地址

- a.next = &b	
让节点 a 指向节点 b

- p->data	
访问指针 p 所指节点的数据

- "%d -> " 是 printf 的输出格式字符串，可以拆成两部分：
%d：占位符，表示这里输出一个整数。
->：普通文字，会原样输出空格和箭头。


## 双链表
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *prev;
    struct Node *next;
} Node;

int main(void) {
    Node *head = NULL;
    Node *tail = NULL;

    /* 创建第一个节点 */
    Node *p = malloc(sizeof(Node));
    if (p == NULL) {
        return 1;
    }

    p->data = 10;
    p->prev = NULL;
    p->next = NULL;

    head = p;
    tail = p;

    /* 创建第二个节点 */
    Node *q = malloc(sizeof(Node));
    if (q == NULL) {
        free(p);
        return 1;
    }

    q->data = 20;
    q->prev = tail;
    q->next = NULL;

    tail->next = q;
    tail = q;

    /* 从头到尾遍历 */
    for (Node *cur = head; cur != NULL; cur = cur->next) {
        printf("%d ", cur->data);
    }

    printf("\n");

    /* 释放内存 */
    free(q);
    free(p);

    return 0;
}
程序输出：
10 20

- 三部分：
数据域 data
指向前一个节点的指针 prev
向后一个节点的指针 next

- #include <stdio.h>
引入标准输入输出库，使用 printf() 时需要它。

- #include <stdlib.h>
引入标准库，使用 malloc() 和 free() 时需要它。

- typedef struct Node {
}
定义一个结构体，名字是 struct Node，并准备给它起一个简写名称。

- int data;
数据域，用来保存节点中的数据。本程序保存整数。

- struct Node *prev;
指针域，指向当前节点的前一个节点。
这里必须写 struct Node *，因为结构体还没有定义结束，不能直接使用 Node *。

- struct Node *next;
指针域，指向当前节点的后一个节点。
这里必须写 struct Node *，因为结构体还没有定义结束，不能直接使用 Node *。

- {
    } Node;
定义结束，并用 Node 作为 struct Node 的简写。

- 以后可以直接写：Node *p;

- int main(void) {}
    程序入口，程序从这里开始执行。

- Node *head = NULL;
定义头指针 head，初始化为 NULL。
head 指向双链表的第一个节点。刚开始链表为空，所以没有节点可指向。

- Node *tail = NULL;
定义尾指针 tail，指向双链表的最后一个节点。
刚开始链表为空，所以也是 NULL。

- 此时链表状态为：
head -> NULL
tail -> NULL

- Node *p = malloc(sizeof(Node));
动态申请一块内存，用来存放一个 Node 类型的节点。
malloc() 返回的是一块内存的地址，所以用 Node *p 保存。
sizeof(Node) 表示一个节点所需要的字节数。

- if (p == NULL) {
    return 1;
}
判断内存申请是否失败。
如果内存不足，`malloc()` 会返回 NULL。
如果申请失败，直接结束程序。返回 1 通常表示程序异常结束。

-  p->data = 10;
把节点 p 的数据域设置为 10。

- 因为 p 是第一个节点，前面没有节点，所以：
p->prev = NULL;

- 此时还没有后继节点，所以：
p->next = NULL;

- 现在节点 p 的结构是：
NULL <- [10] -> NULL

- head = p;
让头指针指向节点 p。

- tail = p;
因为当前链表中只有一个节点，所以头节点和尾节点都是 p。

- Node *q = malloc(sizeof(Node));
再次动态申请一块内存，用来存储第二个节点，并让指针 q 指向它。

-  if (q == NULL) {
    free(p);
    return 1;
} 
判断第二次内存申请是否失败。如果申请失败，先释放之前申请的节点 p，避免内存泄漏。

- q->data = 20;
将第二个节点的数据设置为 20。

- q->prev = tail;
让 q 的前指针指向原来的尾节点。

- q->next = NULL;
因为 q 暂时是最后一个节点，所以它的后指针为 NULL。

- tail->next = q;
让原来的尾节点 [10] 的 next 指针指向新节点 [20]。

- tail = q;
更新尾指针，让 tail 指向新的最后一个节点 [20]。

- NULL <- [10] <-> [20] -> NULL
           ^         ^
          head      tail

- for (Node *cur = head; cur != NULL; cur = cur->next) {}
从头节点开始遍历链表。
Node *cur = head;  定义指针 cur，让它从头节点开始。
cur != NULL;  只要 cur 不为空，就继续循环。
cur = cur->next;  每循环一次，`cur` 移动到下一个节点。

- printf("\n");
输出换行符，让光标移动到下一行。

- return 0;
程序正常结束。
返回 0 通常表示程序执行成功。
